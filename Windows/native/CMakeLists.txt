cmake_minimum_required(VERSION 3.20)
project(BlahWindows LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_AVX OFF CACHE BOOL "" FORCE)
set(GGML_AVX2 OFF CACHE BOOL "" FORCE)
set(GGML_FMA OFF CACHE BOOL "" FORCE)
set(GGML_F16C OFF CACHE BOOL "" FORCE)
set(GGML_AVX512 OFF CACHE BOOL "" FORCE)
set(GGML_SSE42 OFF CACHE BOOL "" FORCE)
set(GGML_BMI2 OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_CUDA OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
if(BLAH_ENGINE STREQUAL "speech")
  set(TRANSCRIBE_BUILD_SHARED OFF CACHE BOOL "" FORCE)
  set(TRANSCRIBE_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(TRANSCRIBE_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(TRANSCRIBE_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
  set(TRANSCRIBE_USE_SYSTEM_BLAS OFF CACHE BOOL "" FORCE)
  set(TRANSCRIBE_USE_OPENMP OFF CACHE BOOL "" FORCE)
  set(TRANSCRIBE_X86_CONSERVATIVE ON CACHE BOOL "" FORCE)
  add_subdirectory("${BLAH_VENDOR}/transcribe-cpp-sys-0.1.3" transcribe)
  # This release's subproject uses CMAKE_SOURCE_DIR for its public include path.
  target_include_directories(transcribe PUBLIC "${BLAH_VENDOR}/transcribe-cpp-sys-0.1.3/include")
  add_library(blah_speech SHARED speech.cpp)
  target_include_directories(blah_speech PRIVATE "${BLAH_VENDOR}/transcribe-cpp-sys-0.1.3/include")
  target_link_libraries(blah_speech PRIVATE transcribe)
  set(blah_target blah_speech)
elseif(BLAH_ENGINE STREQUAL "cleanup")
  set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
  set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
  set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
  set(LLAMA_BUILD_APP OFF CACHE BOOL "" FORCE)
  add_subdirectory("${BLAH_VENDOR}/llama-cpp-sys-2-0.1.154/llama.cpp" llama)
  add_library(blah_cleanup SHARED cleanup.cpp)
  target_link_libraries(blah_cleanup PRIVATE llama)
  set(blah_target blah_cleanup)
else()
  message(FATAL_ERROR "Set BLAH_ENGINE to speech or cleanup")
endif()
set_target_properties(${blah_target} PROPERTIES PREFIX "" RUNTIME_OUTPUT_DIRECTORY "${BLAH_OUTPUT}")
foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
  set_target_properties(${blah_target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${config} "${BLAH_OUTPUT}")
endforeach()
if(MINGW)
  target_link_options(${blah_target} PRIVATE -static-libgcc -static-libstdc++)
endif()
